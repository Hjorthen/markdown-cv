\def\InsertImageAtSection#1{
    $if(portrait_image)$
     \doifsamestring\currentstructuretitle{#1}{
	     	      \message { Inserting portrait image }
	              \placefigure[right, none]{}{\blank[-2*big]\externalfigure[$portrait_image$][frame=off, height=5cm]}
     }    
    $endif$
}

\usesymbols[mis]

\def\NodeBox#1{
        \hbox{
           {\lower 2pt \hbox {\hskip 6pt \tfc \symbol[bullet] \hskip -6pt}}
           \framed[topframe=off, bottomframe=off,loffset=4pt, roffset=4pt, location=depth, corner=rectangular, background=color, backgroundcolor=white]{#1} 
           \hskip -5.5pt
           {\raise 2pt \hbox{ \txx \symbol[blacktriangle] }}
        }
}
\define\MakeCompetencies{
        \vskip 14pt
        \hrule height 1pt
        \vskip -12pt
        \dontleavehmode\hbox to \hsize{%
                \hfill
                $for(competences)$ 
                    \NodeBox{$it$}
                    \hfill
                $endfor$
        }
}
